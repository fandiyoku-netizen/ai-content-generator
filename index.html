async function triggerTask(taskName) { const statusBox = document.getElementById('status-box'); statusBox.innerHTML = `[Connecting...] Mengirim perintah ${taskName} ke LangGraph Agent...`; try { const response = await fetch('https://url-backend-python-anda.com/api/run-automation', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ task: taskName }) }); const result = await response.json(); statusBox.innerHTML = `[Success] ${result.message}`; } catch (error) { statusBox.innerHTML = `[Error] Gagal terhubung ke backend AI.`; } }
